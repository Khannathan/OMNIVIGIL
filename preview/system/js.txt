function placeCities(want, main) {
  const out = []; let tries = 0;
  const rnd = (i) => h01(i, 77, SEED & 0xffff, SEED);
  while (out.length < want && tries < 4000) {
    tries++;
    const u = rnd(tries*2)*2-1, ph = rnd(tries*2+1)*Math.PI*2, r = Math.sqrt(1-u*u);
    const dn = V.norm([r*Math.cos(ph), u*0.85, r*Math.sin(ph)]);
    if (BIOME === 13) { if (Math.abs(dn[1]) > 0.6 || out.some(c => V.dot(c.d, dn) > 0.985)) continue;
      out.push({ d: dn, r: out.length ? 0.0012 + rnd(tries*3)*0.0012 : 0.0026, name: CITY_NAMES[(out.length + (SEED % 7)) % CITY_NAMES.length] }); continue; }
    const h = H(dn[0], dn[1], dn[2], 9);
    if (BIOME === 8) { const lc = landClass(dn[0], dn[1], dn[2], 9); if (lc === 'water' || lc === 'snow') continue; }
    if (h < (out.length ? 0.03 : 0.006) || h > 0.3 || Math.abs(dn[1]) > 0.75) continue;
    if (out.some(c => V.dot(c.d, dn) > 0.985)) continue;
    const reach = out.length === 0 ? 0.008 : 0.03;
    let coast = false; for (let a = 0; a < 12 && !coast; a++) { const e = V.norm(V.add(dn, [Math.cos(a)*reach, Math.sin(a*1.7)*reach, Math.sin(a)*reach])); if (H(e[0], e[1], e[2], 9) < 0) coast = true; }
    if (!coast && out.length < 6 && tries < 3000) continue;
    const big = out.length === 0;
    out.push({ d: dn, r: big ? (main ? 0.0042 : 0.0034) : 0.0016 + rnd(tries*3)*0.0018, name: CITY_NAMES[(out.length + (SEED % 7)) % CITY_NAMES.length] });
  }
  const c0 = out[0];
  if (c0 && BIOME !== 13) {
    let best = null;
    for (let a = 0; a < 24 && !best; a++) for (let k = 4; k <= 30; k++) {
      const ang = a/24*Math.PI*2; const north = V.norm(V.sub([0,1,0], V.mul(c0.d, c0.d[1]))); const east = V.norm(V.cross(north, c0.d));
      const p = V.norm(V.add(c0.d, V.add(V.mul(east, Math.cos(ang)*c0.r*k*0.1), V.mul(north, Math.sin(ang)*c0.r*k*0.1))));
      if (H(p[0], p[1], p[2], 12) < 0) { if (!best || k < best.k) best = { k, ang }; break; }
    }
    const ang = best ? best.ang : 0; let north = V.norm(V.sub([0,1,0], V.mul(c0.d, c0.d[1]))); let east = V.norm(V.cross(north, c0.d));
    if (best) {
      const dirw = V.add(V.mul(east, Math.cos(ang)), V.mul(north, Math.sin(ang)));
      const dw = best.k*0.1*c0.r; c0.d = V.norm(V.add(c0.d, V.mul(dirw, Math.max(0, dw - c0.r*0.95))));
      north = V.norm(V.sub([0,1,0], V.mul(c0.d, c0.d[1]))); east = V.norm(V.cross(north, c0.d));
    }
    c0.edge = V.norm(V.add(c0.d, V.add(V.mul(east, Math.cos(ang)*c0.r*0.82), V.mul(north, Math.sin(ang)*c0.r*0.82))));
    c0.hd = Math.PI/2 - ang;
  }
  return out;
}

// Body regime (D <= DB): orbit and ground views around a surface point T, as before.
// System regime (D >= DS): look down on the ecliptic at a free point Q from a fixed tilt.
// Between them the target, view direction and up vector blend, so the zoom never cuts.
const DB = 3.4, DS = 14, TILT = 52*Math.PI/180, DMAXW = 520;
const cam = { T: V.norm([0.35, 0.42, 0.84]), D: 3.2, hd: 0, Q: [0,0,0], sh: 0, fov: 30 };
const clamp01 = x => Math.min(1, Math.max(0, x));
function bodyK(D) { const x = clamp01((Math.log(D) - Math.log(DB)) / (Math.log(DS) - Math.log(DB))); return 1 - x*x*(3-2*x); }
function sysDir(sh) { return [Math.sin(TILT)*Math.sin(sh), Math.cos(TILT), Math.sin(TILT)*Math.cos(sh)]; }
const lerp3 = (a, b, t) => [a[0]+(b[0]-a[0])*t, a[1]+(b[1]-a[1])*t, a[2]+(b[2]-a[2])*t];
let PX = 2;
function frame() {
  const T = cam.T, D = cam.D, kb = bodyK(D);
  const lg = Math.log10(D);
  const k = clamp01((-0.35 - lg) / 1.75); const kk = k*k*(3-2*k);
  const pitch = kk * 52 * Math.PI / 180;
  let north = V.sub([0,1,0], V.mul(T, T[1])); if (Math.hypot(...north) < 1e-6) north = [0,0,1]; north = V.norm(north);
  const east = V.norm(V.cross(north, T));
  const fh = V.add(V.mul(north, Math.cos(cam.hd)), V.mul(east, Math.sin(cam.hd)));
  const offB = V.add(V.mul(T, Math.cos(pitch)), V.mul(fh, -Math.sin(pitch)));
  const fwdB = V.mul(offB, -1), rightB = V.norm(V.cross(fh, T)), upB = V.norm(V.cross(rightB, fwdB));
  const S = sysDir(cam.sh), upS = V.norm(V.sub([0,1,0], V.mul(S, S[1])));
  const off = kb >= 1 ? offB : V.norm(lerp3(S, offB, kb));
  const upG = kb >= 1 ? upB : V.norm(lerp3(upS, upB, kb));
  const target = kb >= 1 ? T : lerp3(cam.Q, T, kb);
  const pos = V.add(target, V.mul(off, D));
  const fwd = V.mul(off, -1);
  const right = V.norm(V.cross(fwd, upG)); const up = V.norm(V.cross(right, fwd));
  return { pos, fwd, right, up, fh, pitch, T, kb, target };
}
function rayDir(F, sx, sy) { const th = Math.tan(cam.fov*Math.PI/360), asp = cv.width / cv.height;
  return V.norm(V.add(F.fwd, V.add(V.mul(F.right, sx*th*asp), V.mul(F.up, sy*th)))); }
function hitSphere(o, d) { const b = V.dot(o, d), c = V.dot(o, o) - 1, disc = b*b - c; if (disc < 0) return null; const t = -b - Math.sqrt(disc); if (t < 0) return null; return V.norm(V.add(o, V.mul(d, t))); }
const local = (w) => { const f = bodies[FOCUS]; return V.mul(V.sub(w, f.C), 1/f.R); };   // world -> focus units
let lastKb = 1;
function sync() {
  const kb = bodyK(cam.D);
  if (lastKb >= 1 && kb < 1 && !anim) cam.sh = Math.atan2(cam.T[0], cam.T[2]);
  if (kb <= 0) { cam.T = sysDir(cam.sh); cam.hd = 0; }
  if (kb >= 1) cam.Q = [0,0,0];
  lastKb = kb;
}
function setFocus(i) {
  if (i === FOCUS) return;
  const a = bodies[FOCUS], b = bodies[i];
  const Qw = V.add(a.C, V.mul(cam.Q, a.R));
  cam.Q = V.mul(V.sub(Qw, b.C), 1/b.R); cam.D = cam.D*a.R/b.R;
  FOCUS = i; useBody(i); propKey = ''; refresh();
  if (bodyK(cam.D) <= 0) { cam.T = sysDir(cam.sh); cam.hd = 0; }
}
function refocus() {
  if (bodyK(cam.D) > 0) return;
  const f = bodies[FOCUS], Qw = V.add(f.C, V.mul(cam.Q, f.R)), Dw = cam.D*f.R;
  let best = FOCUS, bs = Infinity;
  for (const b of bodies) { if (Dw/b.R < DS) continue; const d = Math.hypot(...V.sub(Qw, b.C))/b.R; if (d < bs) { bs = d; best = b.i; } }
  setFocus(best);
}

function groundList(F) {
  const all = propList.slice();

  for (const u of units) all.push({ p: V.sub(u.d, instOrigin), frame: [u.frame[0] + tframe*64, u.frame[1], u.frame[2], u.frame[3]], w: u.w, h: u.h, lift: u.hover ? u.hover + Math.sin(performance.now()/400)*3*TEXL : 0 });

  for (const q of all) q.z = V.dot(V.add(rel, q.p), F.fwd);
  all.sort((a, b) => b.z - a.z);
  return all;
}

  const P = bodies[CRAFT.from], M = bodies[CRAFT.to];
  const pd = P.cities[0] ? P.cities[0].d : [0,1,0], md = M.cities[0] ? M.cities[0].d : [0,1,0];
  const p0 = V.add(P.C, V.mul(pd, P.R)), p1 = V.add(P.C, V.mul(pd, P.R*1.5));
  const m0 = V.add(M.C, V.mul(md, M.R)), m1 = V.add(M.C, V.mul(md, M.R*2.2));
  if (s < 0.12) { const e = s/0.12; return V.add(p0, V.mul(V.sub(p1, p0), e*e)); }
  if (s > 0.88) { const e = (1-s)/0.12; return V.add(m0, V.mul(V.sub(m1, m0), e*e)); }
  const e = (s-0.12)/0.76, ee = e*e*(3-2*e);
  return bez(p1, V.add(P.C, V.mul(pd, P.R*3.2)), V.add(M.C, V.mul(md, M.R*6)), m1, ee);
}
function craftState() {
  const u = simT % CRAFT.per, half = CRAFT.per/2, fly = half*0.82;
  let s, parked = false;
  if (u < fly) s = u/fly; else if (u < half) { s = 1; parked = true; }
  else if (u < half + fly) s = 1 - (u-half)/fly; else { s = 0; parked = true; }
  const ds = u < half ? 0.002 : -0.002;
  return { w: craftPath(s), v: parked ? null : V.sub(craftPath(Math.min(1, Math.max(0, s+ds))), craftPath(s)), parked, s };
}
function occluded(F, p) { // is focus-local point p hidden behind a body?
  const d = V.sub(p, F.pos), L = Math.hypot(...d), rd = V.mul(d, 1/L);
  for (const b of bodies) { const c = V.sub(local(b.C), F.pos), R = b.R/bodies[FOCUS].R*0.998;
    const bb = V.dot(c, rd); if (bb <= 0) continue; const cx = V.cross(c, rd); const d2 = V.dot(cx, cx); if (d2 >= R*R) continue;
    if (bb - Math.sqrt(R*R - d2) < L*0.999) return true; }
  const s = V.sub(local([0,0,0]), F.pos), Rs = SUN_R/bodies[FOCUS].R, bs = V.dot(s, rd);
  if (bs > 0 && bs < L) { const cx = V.cross(s, rd); if (V.dot(cx, cx) < Rs*Rs) return true; }
  return false;
}
